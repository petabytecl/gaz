**Method 1: Schedule (lines 136-150)**
Replace with:
```go
func (c *Cron) Schedule(schedule Schedule, cmd Job) EntryID {
    c.runningMu.Lock()
    c.nextID++
    entry := &Entry{
        ID:         c.nextID,
        Schedule:   schedule,
        WrappedJob: c.chain.Then(cmd),
        Job:        cmd,
    }
    if !c.running {
        c.entries = append(c.entries, entry)
        c.runningMu.Unlock()
        return entry.ID
    }
    c.runningMu.Unlock()

    select {
    case c.add <- entry:
    case <-c.stop:
    }
    return entry.ID
}
```

**Method 2: Entries (lines 154-163)**
Replace with:
```go
func (c *Cron) Entries() []Entry {
    c.runningMu.Lock()
    if !c.running {
        snapshot := c.entrySnapshot()
        c.runningMu.Unlock()
        return snapshot
    }
    c.runningMu.Unlock()

    replyChan := make(chan []Entry, 1)
    select {
    case c.snapshot <- replyChan:
        return <-replyChan
    case <-c.stop:
        return nil
    }
}
```

**Method 3: Remove (lines 181-189)**
Replace with:
```go
func (c *Cron) Remove(id EntryID) {
    c.runningMu.Lock()
    if !c.running {
        c.removeEntry(id)
        c.runningMu.Unlock()
        return
    }
    c.runningMu.Unlock()

    select {
    case c.remove <- id:
    case <-c.stop:
    }
}
```

**Method 4: Stop (lines 302-315)**
Replace with:
```go
func (c *Cron) Stop() context.Context {
    c.runningMu.Lock()
    if !c.running {
        c.runningMu.Unlock()
        ctx, cancel := context.WithCancel(context.Background())
        cancel()
        return ctx
    }
    c.running = false
    c.runningMu.Unlock()

    select {
    case c.stop <- struct{}{}:
    default:
        // Scheduler already exited
    }

    ctx, cancel := context.WithCancel(context.Background())
    go func() {
        c.jobWaiter.Wait()
        cancel()
    }()
    return ctx
}
```

Note for Stop: The original code set `c.running = false` under lock, which is correct. But it then sent on `c.stop` while holding the lock. The fix keeps the `c.running = false` under lock (necessary for the state transition) but sends on the channel outside the lock. Use `select { case c.stop <- struct{}{}: default: }` to handle the case where the scheduler goroutine has already exited.

Also note: The `Start()` and `Run()` methods (lines 192-212) use `defer c.runningMu.Unlock()` but do NOT send on channels under lock -- they only set `c.running = true` and call `go c.run()`. These are correct and do not need changes.

**Step 2: Refactor lazySingleton.GetInstance (lines 151-172).**

Replace with:
```go
func (s *lazySingleton[T]) GetInstance(c *Container, chain []string) (any, error) {
    // Fast path: already built
    if s.built.Load() {
        return s.instance, nil
    }

    // Re-entrant cycle detection: if the same goroutine is already initializing this
    // singleton (via a dependency chain), fail fast instead of deadlocking on Once.Do.
    currentGoid := getGoroutineID()
    if initGoid := s.initGoid.Load(); initGoid != 0 && initGoid == currentGoid {
        return nil, fmt.Errorf("%w: re-entrant resolution of %s", ErrCycle, s.serviceName)
    }

    s.once.Do(func() {
        s.initGoid.Store(getGoroutineID())
        defer s.initGoid.Store(0)

        instance, err := s.provider(c)
        if err != nil {
            s.err = err
            return
        }

        if injectErr := injectStruct(c, instance, chain); injectErr != nil {
            s.err = injectErr
            return
        }

        s.instance = instance
        s.built.Store(true)
    })

    return s.instance, s.err
}
```
