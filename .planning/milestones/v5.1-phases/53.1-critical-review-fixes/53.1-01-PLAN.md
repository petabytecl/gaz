    if err := a.Build(); err != nil {
        return err
    }

        a.mu.Unlock()
        return errors.New("app is already running")

}

**Step 2: Fix Publish to use snapshot-under-lock pattern.**

Replace the current Publish function (lines 154-191) with:
```go
func Publish[T Event](ctx context.Context, b *EventBus, event T, topic string) {
    b.mu.RLock()
    if b.closed {
        b.mu.RUnlock()
        return
    }

    eventType := reflect.TypeOf(event)

    // Snapshot handlers under lock
    var handlers []*asyncSubscription
    exactKey := subscriptionKey{eventType: eventType, topic: topic}
    handlers = append(handlers, b.handlers[exactKey]...)
    if topic != "" {
        wildcardKey := subscriptionKey{eventType: eventType, topic: ""}
        handlers = append(handlers, b.handlers[wildcardKey]...)
    }
    b.mu.RUnlock()

    // Deliver OUTSIDE lock -- no deadlock possible
    env := eventEnvelope{ctx: ctx, event: event}
    for _, h := range handlers {
        select {
        case h.ch <- env:
            // Delivered
        case <-h.tombstone:
            // Subscription cancelled during send -- skip
        case <-ctx.Done():
            return
        }
    }
}
```

Remove the old comment at line 176-178 that codified the bug as intentional.

**Step 3: Fix Close to set tombstones before closing channels.**

Replace the current Close function (lines 222-251) with:
```go
func (b *EventBus) Close() {
    b.mu.Lock()
    if b.closed {
        b.mu.Unlock()
        return
    }
    b.closed = true

    // Collect all subscriptions
    var allSubs []*asyncSubscription
    for _, subs := range b.handlers {
        allSubs = append(allSubs, subs...)
    }

    // Signal tombstones first -- in-flight Publish senders will short-circuit
    for _, sub := range allSubs {
        close(sub.tombstone)
    }

    // Close channels to drain handler goroutines
    for _, sub := range allSubs {
        close(sub.ch)
    }
    b.mu.Unlock()

    // Wait for handlers outside lock
    for _, sub := range allSubs {
        <-sub.done
    }

    b.logger.Info("eventbus stopped", "subscriptions_drained", len(allSubs))
}
```

**Step 4: Fix unsubscribe to release lock before waiting on sub.done.**

Replace the current unsubscribe function (lines 258-281) with:
```go
func (b *EventBus) unsubscribe(eventType reflect.Type, topic string, id uint64) {
    b.mu.Lock()

    if b.closed {
        b.mu.Unlock()
        return
    }

    key := subscriptionKey{eventType: eventType, topic: topic}
    subs := b.handlers[key]

    var target *asyncSubscription
    for i, sub := range subs {
        if sub.id == id {
            target = sub
            // Remove from map under lock
            b.handlers[key] = append(subs[:i], subs[i+1:]...)
            if len(b.handlers[key]) == 0 {
                delete(b.handlers, key)
            }
            break
        }
    }

    if target == nil {
        b.mu.Unlock()
        return
    }

    // Signal and close under lock to prevent send-on-closed-channel
    close(target.tombstone)
    close(target.ch)
    b.mu.Unlock()

    // Wait for handler goroutine outside lock
    <-target.done
}
```
